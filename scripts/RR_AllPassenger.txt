// Rail Reverse - all passenger trains: on / off.
// The plugin switches the passenger trains on while RR_AllPassengerOn.txt is running;
// this button starts or stops it. The running script is kept in the save.

include("SOVIETInstructions.txt");

defineVariable(char, bOn);

defineFunction(main, void)
{
	Script_Exists("RR_AllPassengerOn.txt", bOn);
	if (bOn) { Script_DestroyByName("RR_AllPassengerOn.txt"); }
	if (!bOn) { Script_StartNew("RR_AllPassengerOn.txt"); }
	end();
}
